tutorial tailwind : 


# 1. Initialize project with package.json
npm init -y

# 2. Install Tailwind v4 and the official CLI
npm install tailwindcss @tailwindcss/cli

src/input.css : 
  @import "tailwindcss";

package.json : 
  "scripts": {
  "dev": "tailwindcss -i ./src/input.css -o ./src/output.css --watch",
  "build": "tailwindcss -i ./src/input.css -o ./src/output.css --minify"
  }
